// WeldControl Studio · Liburdi Automation · Rev 4.2
public class MotionController
{
    private readonly IPlcDriver _plc;
    private readonly IWeldLogger _log;
    private volatile bool _eStop = false;

    public async Task ExecuteProgramAsync(WeldProgram prog)
    {
        await _plc.SetTravelSpeed(prog.Parameters.TravelSpeed);
        await _plc.SetArcVoltage(prog.Parameters.ArcVoltage);

        foreach (var step in prog.Sequence)
        {
            if (_eStop) throw new EStopException();
            _log.Info($"Step {step.N}: {step.Description}");
            await RunStepAsync(step);
            await ValidateWeldQuality(step);
        }
    }

    private async Task ValidateWeldQuality(ProgramStep step)
    {
        var r = await _plc.GetSensorReading();
        if (r.Voltage < 11.5 || r.Voltage > 14.0)
            throw new WeldFaultException("Arc voltage OOB", r);
    }
}
